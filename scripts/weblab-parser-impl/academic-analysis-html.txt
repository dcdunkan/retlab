<script type="1feef66dd8bfa2b7c38b0463-text/javascript">
(function() {
	function ensureModalContainer() {
		var modal = document.getElementById('attendanceRequestModal');
		if (!modal) {
			modal = document.createElement('div');
			modal.id = 'attendanceRequestModal';
			modal.className = 'modal hide fade';
			modal.tabIndex = -1;
			modal.setAttribute('role', 'dialog');
			modal.setAttribute('aria-hidden', 'true');
			modal.style.cssText = 'width: 750px; margin-left: -375px; z-index: 10500; display: none; background: #fff; border-radius: 6px; border: 1px solid #ccc; box-shadow: 0 5px 15px rgba(0,0,0,0.5); position: fixed; top: 10%; left: 50%;';
			modal.innerHTML = '<div class="modal-header" style="padding: 9px 15px; border-bottom: 1px solid #eee; background: #f5f5f5;"><button type="button" class="close" data-dismiss="modal" aria-hidden="true" style="float: right; font-size: 20px; font-weight: bold; line-height: 20px; color: #000; opacity: .4; border: 0; background: transparent; cursor: pointer;">&times;</button><h3 id="attendanceRequestModalLabel" style="margin: 0; line-height: 30px; font-size: 18px;">Send Attendance Request</h3></div><div class="modal-body" id="attendanceRequestModalBody" style="max-height: 500px; padding: 15px; overflow-y: auto;"><div style="text-align: center; padding: 20px;">Loading...</div></div>';
			document.body.appendChild(modal);
		}
		return modal;
	}

	function showModal(requestUrl) {
		var modal = ensureModalContainer();
		var modalBody = document.getElementById('attendanceRequestModalBody');
		if (!modalBody) return;

		modalBody.setAttribute('data-url', requestUrl);
		modalBody.innerHTML = '<div style="text-align: center; padding: 30px;"><img src="/images/ajax-loader.gif" alt="Loading..."/> Loading request form...</div>';

		if (typeof window.jQuery !== 'undefined' && typeof window.jQuery.fn.modal === 'function') {
			window.jQuery(modal).modal('show');
		} else {
			modal.style.display = 'block';
			modal.classList.add('in');
			var backdrop = document.getElementById('attendanceRequestModalBackdrop');
			if (!backdrop) {
				backdrop = document.createElement('div');
				backdrop.id = 'attendanceRequestModalBackdrop';
				backdrop.className = 'modal-backdrop fade in';
				backdrop.style.cssText = 'position: fixed; top: 0; right: 0; bottom: 0; left: 0; z-index: 10400; background-color: #000; opacity: 0.5;';
				document.body.appendChild(backdrop);
			} else {
				backdrop.style.display = 'block';
			}
		}

		var xhr = new XMLHttpRequest();
		var separator = requestUrl.indexOf('?') !== -1 ? '&' : '?';
		var urlWithAjax = requestUrl + separator + 'ajax=1';
		xhr.open('GET', urlWithAjax, true);
		xhr.setRequestHeader('X-Requested-With', 'XMLHttpRequest');
		xhr.onload = function() {
			if (xhr.status >= 200 && xhr.status < 400) {
				// Strip any stylesheet link tags to prevent main page sidebar/layout reflow distortion
				var cleanHtml = xhr.responseText.replace(/<link[^>]*rel=["']stylesheet["'][^>]*>/gi, '');
				modalBody.innerHTML = cleanHtml;
				var scripts = modalBody.getElementsByTagName('script');
				for (var i = 0; i < scripts.length; i++) {
					try {
						eval(scripts[i].innerText || scripts[i].textContent);
					} catch (err) {}
				}
			} else {
				modalBody.innerHTML = '<div class="alert alert-error">Failed to load request content.</div>';
			}
		};
		xhr.onerror = function() {
			modalBody.innerHTML = '<div class="alert alert-error">Network error. Failed to load content.</div>';
		};
		xhr.send();
	}

	function closeModal() {
		var modal = document.getElementById('attendanceRequestModal');
		if (modal) {
			if (typeof window.jQuery !== 'undefined' && typeof window.jQuery.fn.modal === 'function') {
				window.jQuery(modal).modal('hide');
			}
			modal.style.display = 'none';
			modal.classList.remove('in');
		}
		var backdrop = document.getElementById('attendanceRequestModalBackdrop');
		if (backdrop) {
			backdrop.style.display = 'none';
		}
	}

	document.addEventListener('click', function(e) {
		var target = e.target;
		if (!target) return;

		if (target.matches('[data-dismiss="modal"], .close, #attendanceRequestModalBackdrop')) {
			closeModal();
			return;
		}

		while (target && target.tagName !== 'A') {
			target = target.parentElement;
		}
		if (!target) return;

		var href = target.getAttribute('href');
		if (!href || href === '#' || href.indexOf('javascript:') === 0) return;

		var lowerHref = href.toLowerCase();
		if (lowerHref.indexOf('sendpermissionrequest') !== -1 || (lowerHref.indexOf('sendrequest') !== -1 && lowerHref.indexOf('viewrequest') === -1 && lowerHref.indexOf('approverequest') === -1)) {
			e.preventDefault();
			showModal(href);
		}
	}, true);

	document.addEventListener('submit', function(e) {
		var form = e.target;
		var modalBody = document.getElementById('attendanceRequestModalBody');
		if (!modalBody || !modalBody.contains(form)) return;

		e.preventDefault();
		var actionUrl = form.getAttribute('action') || modalBody.getAttribute('data-url') || location.href;
		
		var formData = new FormData(form);
		formData.append('ajax', '1');

		var oldMsg = document.getElementById('modal-submit-msg');
		if (oldMsg) oldMsg.parentNode.removeChild(oldMsg);

		var infoDiv = document.createElement('div');
		infoDiv.id = 'modal-submit-msg';
		infoDiv.className = 'alert alert-info';
		infoDiv.textContent = 'Submitting request...';
		modalBody.insertBefore(infoDiv, modalBody.firstChild);

		var xhr = new XMLHttpRequest();
		xhr.open('POST', actionUrl, true);
		xhr.setRequestHeader('X-Requested-With', 'XMLHttpRequest');
		xhr.onload = function() {
			var msgDiv = document.getElementById('modal-submit-msg');
			if (msgDiv) msgDiv.parentNode.removeChild(msgDiv);

			var isJson = false;
			var data = null;
			try {
				data = JSON.parse(xhr.responseText);
				isJson = true;
			} catch(err) {}

			if (isJson && data) {
				if (data.status === 'success' || data.success) {
					modalBody.innerHTML = '<div class="alert alert-success">' + (data.message || 'Request submitted successfully.') + '</div>';
					setTimeout(function() {
						closeModal();
						window.location.reload();
					}, 1500);
				} else if (data.status === 'error' || data.error || data.status === false) {
					var errDiv = document.createElement('div');
					errDiv.id = 'modal-submit-msg';
					errDiv.className = 'alert alert-error';
					errDiv.textContent = data.message || data.error || 'Failed to process request.';
					modalBody.insertBefore(errDiv, modalBody.firstChild);
				} else {
					var cleanResponse = xhr.responseText.replace(/<link[^>]*rel=["']stylesheet["'][^>]*>/gi, '');
					modalBody.innerHTML = cleanResponse;
				}
			} else {
				var cleanResponse = xhr.responseText.replace(/<link[^>]*rel=["']stylesheet["'][^>]*>/gi, '');
				modalBody.innerHTML = cleanResponse;
			}
		};
		xhr.onerror = function() {
			var msgDiv = document.getElementById('modal-submit-msg');
			if (msgDiv) msgDiv.parentNode.removeChild(msgDiv);
			alert('Error processing request.');
		};
		xhr.send(formData);
	}, true);
})();
</script>

$(document).ready(function () {
    $('input').keyup(function (e) {
    if (e.which == 39) { // right arrow
      $(this).closest('td').next().find('input').focus();

    } else if (e.which == 37) { // left arrow
      $(this).closest('td').prev().find('input').focus();

    } else if (e.which == 40) { // down arrow
      $(this).closest('tr').next().find('td:eq(' + $(this).closest('td').index() + ')').find('input').focus();

    } else if (e.which == 38) { // up arrow
      $(this).closest('tr').prev().find('td:eq(' + $(this).closest('td').index() + ')').find('input').focus();
    }
    });
});
